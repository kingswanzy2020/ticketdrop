+# true leaves replicas out of the Deployment, so the HPA that KEDA creates
+# owns the count and an upgrade does not reset it. The ScaledObject that
+# drives it comes in step 10 task 4. replicaCount is then not rendered, but
+# pdb.yaml still reads it: keep it at 1 and the service gets no budget.
+autoscaling:
+  enabled: false
+  # How KEDA's operator signs its calls to the queue (entry 54). Not the
+  # service's own credentials: on AWS the two are different roles.
+  auth:
+    # secret: a Secret <release>-keda-aws with the two keys below, for goaws,
+    #   which accepts anything but the scaler insists on something.
+    # aws: pod identity, an IAM role on EKS; no Secret is rendered.
+    provider: secret
+    # Placeholders only. A real credential never goes in values.
+    accessKeyID: local
+    secretAccessKey: local

$ cat charts/service/templates/keda-aws-secret.yaml charts/service/templates/triggerauthentication.yaml
{{- /* The scaler's placeholder credentials for goaws (entry 54). */}}
{{- $auth := .Values.autoscaling.auth }}
{{- if and .Values.autoscaling.enabled (eq $auth.provider "secret") }}
apiVersion: v1
kind: Secret
metadata:
  name: {{ include "service.name" . }}-keda-aws
  labels:
    {{- include "service.labels" . | nindent 4 }}
type: Opaque
stringData:
  AWS_ACCESS_KEY_ID: {{ required "autoscaling.auth.accessKeyID is needed with provider secret" $auth.accessKeyID | quote }}
  AWS_SECRET_ACCESS_KEY: {{ required "autoscaling.auth.secretAccessKey is needed with provider secret" $auth.secretAccessKey | quote }}
{{- end }}
{{- /*
How KEDA's operator authenticates when it reads this service's queue depth.
It must sit in the ScaledObject's namespace (entry 54).
*/}}
{{- if .Values.autoscaling.enabled }}
{{- $provider := .Values.autoscaling.auth.provider }}
{{- if not (has $provider (list "secret" "aws")) }}
{{- fail (printf "autoscaling.auth.provider is %q; use secret or aws" $provider) }}
{{- end }}
apiVersion: keda.sh/v1alpha1
kind: TriggerAuthentication
metadata:
  name: {{ include "service.name" . }}
  labels:
    {{- include "service.labels" . | nindent 4 }}
spec:
  {{- if eq $provider "aws" }}
  podIdentity:
    provider: aws
  {{- else }}
  secretTargetRef:
    - parameter: awsAccessKeyID
      name: {{ include "service.name" . }}-keda-aws
      key: AWS_ACCESS_KEY_ID
    - parameter: awsSecretAccessKey
      name: {{ include "service.name" . }}-keda-aws
      key: AWS_SECRET_ACCESS_KEY
  {{- end }}
{{- end }}
